def _otf_to_ttf(src, text, dst):
    """Noto CJK(OTF/TTC, CFF 윤곽선)를 사용 글자만 남겨 TrueType으로 변환 (reportlab은 CFF 글꼴을 못 씀)"""
    from fontTools.ttLib import TTFont as FT, newTable
    from fontTools import subset
    from fontTools.pens.cu2quPen import Cu2QuPen
    from fontTools.pens.ttGlyphPen import TTGlyphPen
    kw = {"fontNumber": 0}
    if src.endswith(".ttc"):  # TTC 안에서 KR 서체 찾기
        from fontTools.ttLib import TTCollection
        for i, f in enumerate(TTCollection(src).fonts):
            if "KR" in f["name"].getDebugName(1): kw["fontNumber"] = i; break
    font = FT(src, **kw)
    opt = subset.Options(); opt.layout_features = []; opt.name_IDs = ["*"]; opt.notdef_outline = True
    sub = subset.Subsetter(opt); sub.populate(text=text); sub.subset(font)
    gs = font.getGlyphSet(); glyf = newTable("glyf"); glyf.glyphs = {}; glyf.glyphOrder = font.getGlyphOrder()
    for g in font.getGlyphOrder():
        pen = TTGlyphPen(gs); gs[g].draw(Cu2QuPen(pen, 1.0, reverse_direction=True)); glyf[g] = pen.glyph()
    font["loca"] = newTable("loca"); font["glyf"] = glyf
    font["maxp"].tableVersion = 0x00010000
    for k in ["maxZones","maxTwilightPoints","maxStorage","maxFunctionDefs","maxInstructionDefs","maxStackElements",
              "maxSizeOfInstructions","maxComponentElements","maxPoints","maxContours","maxCompositePoints",
              "maxCompositeContours","maxComponentDepth"]:
        setattr(font["maxp"], k, 0)
    font["maxp"].maxZones = 1
    font["head"].glyphDataFormat = 0
    if "post" in font: font["post"].formatType = 3.0
    del font["CFF "]
    if "VORG" in font: del font["VORG"]
    font.sfntVersion = "\x00\x01\x00\x00"; font.save(dst); return dst

_N = [0]
def setup_font(text=""):
    """한글 글꼴을 PDF에 '내장'한다. 1) TTF 한글 글꼴 → 2) Noto CJK 변환 → 실패 시 RuntimeError"""
    ttf = [p for pat in FONT_TTF_PATTERNS for p in glob.glob(pat, recursive=True)]
    reg = [p for p in ttf if "Bold" not in p and "bd" not in os.path.basename(p).lower()] or ttf
    bold = [p for p in ttf if "Bold" in p or "bd" in os.path.basename(p).lower()]
    for p in reg:
        try:
            pdfmetrics.registerFont(TTFont("KR", p))
            pdfmetrics.registerFont(TTFont("KRB", bold[0] if bold else p)); return "KR", "KRB"
        except Exception: continue
    _N[0] += 1; a, bn = f"KR{_N[0]}", f"KRB{_N[0]}"
    otf = [p for pat in FONT_OTF_PATTERNS for p in glob.glob(pat, recursive=True)]
    if otf:
        chars = "".join(sorted(set(text + "0123456789-()[]/.,:%○·•ⅠⅡⅢ①②③ ")))
        r = [p for p in otf if "Bold" not in p][0] if [p for p in otf if "Bold" not in p] else otf[0]
        b = [p for p in otf if "Bold" in p]
        pdfmetrics.registerFont(TTFont(a, _otf_to_ttf(r, chars, f"/tmp/{a}.ttf")))
        pdfmetrics.registerFont(TTFont(bn, _otf_to_ttf(b[0], chars, f"/tmp/{bn}.ttf") if b else f"/tmp/{a}.ttf"))
        return a, bn
    raise RuntimeError("한글 글꼴 없음: NanumGothic.ttf 같은 한글 TTF 글꼴 파일을 첨부한 뒤 다시 실행")

F = FB = None
st = {}
def init(text):
    global F, FB, st
    F, FB = setup_font(text)
    S = lambda name, **kw: ParagraphStyle(name, fontName=kw.pop("font", F), fontSize=kw.pop("size", 10.5),
                                         leading=kw.pop("leading", 16), wordWrap="CJK", **kw)
    st.update({
        "form": S("form", size=9), "title": S("title", font=FB, size=24, leading=30, alignment=TA_CENTER),
        "date": S("date", size=11, alignment=TA_CENTER), "cell": S("cell"), "cellc": S("cellc", alignment=TA_CENTER),
        "h1": S("h1", keepWithNext=1, font=FB, size=14, leading=20, spaceBefore=10, spaceAfter=6),
        "h2": S("h2", keepWithNext=1, font=FB, size=12, leading=18, spaceBefore=8, spaceAfter=4),
        "h3": S("h3", keepWithNext=1, font=FB, size=11, leading=17, spaceBefore=6, spaceAfter=2),
        "h4": S("h4", keepWithNext=1, font=FB, size=10.5, leading=16, spaceBefore=4, leftIndent=8),
        "body": S("body", leftIndent=8, spaceAfter=3), "bul": S("bul", leftIndent=18, firstLineIndent=-8, spaceAfter=2),
        "toc": S("toc", size=11, leading=17),
        "toc0": S("toc0", font=FB, size=12.5, leading=24, spaceBefore=4),
        "lab": S("lab", font=FB, size=10, leading=15, spaceBefore=4, leftIndent=8, textColor=colors.HexColor("#0F6B69")),
        "cap": S("cap", size=9.5, leading=14, alignment=TA_CENTER),
    })

esc = lambda t: str(t).replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;").replace("\n", "<br/>")
P = lambda t, s="body": Paragraph(esc(t), st[s])
def bullets(items, mark="-"): return [P(f"{mark} {x}", "bul") for x in items]

IMG_KEYS = ["문제해결경험", "현상", "원인", "개선방안", "기대효과", "실무비교", "소감"]

def _prep_image(path, i):
    """사진 회전(EXIF) 보정·크기 축소 후 임시 JPG로 저장. 실패하면 원본 경로 사용"""
    try:
        from PIL import Image as PI, ImageOps
        im = ImageOps.exif_transpose(PI.open(path)).convert("RGB"); im.thumbnail((1600, 1600))
        out = f"/tmp/report_img_{i}.jpg"; im.save(out, "JPEG", quality=85); return out, im.size
    except Exception:
        from reportlab.lib.utils import ImageReader
        return path, ImageReader(path).getSize()

def figures(d, key, cnt):
    """위치(key)에 해당하는 이미지들을 [그림 n] 캡션과 함께 반환"""
    out = []
    for i, im in enumerate(d.get("이미지") or []):
        loc = str(im.get("위치", "")).strip()
        valid = loc in IMG_KEYS or loc.startswith("상세:")
        if (loc if valid else "문제해결경험") != key: continue
        if not os.path.exists(im.get("파일", "")): print("이미지 없음:", im.get("파일")); continue
        path, (w, h) = _prep_image(im["파일"], i)
        sc = min(130 * mm / w, 85 * mm / h); cnt[0] += 1
        out.append(KeepTogether([Spacer(1, 4), Image(path, w * sc, h * sc),
                   P(f"[그림 {cnt[0]}] {im.get('캡션', '')}", "cap"), Spacer(1, 4)]))
    return out

            ("1) 현상  2) 원인  3) 개선 방안  4) 기대 효과", 2), ("2. 실무와의 비교", 1),
            ("Ⅲ. 소감 및 계획", 0), ("1. 소감", 1), ("2. 향후 계획", 1)]
    return [P("[서식5] 최종보고서", "form"), Spacer(1, 8), P("최종보고서", "title"), Spacer(1, 6),
            P(d["작성일"], "date"), Spacer(1, 12), info, Spacer(1, 12), box, PageBreak(),
            P("- 목 차 -", "title"), Spacer(1, 14)] + [Paragraph(esc(t), st["toc0" if lv == 0 else "toc"], ) if lv == 0 else Paragraph(esc(t), ParagraphStyle("t%d" % lv, parent=st["toc"], leftIndent=14 * lv, fontSize=11 if lv < 3 else 10, leading=17 if lv < 3 else 15)) for t, lv in toc] + [PageBreak()]

def footer(c, doc):
    c.saveState(); c.setFont(F, 9); c.drawCentredString(A4[0] / 2, 12 * mm, f"- {doc.page} -"); c.restoreState()

def build(d, out):
    if d.get("교과목키"):
        d = {**MODULES[d["교과목키"]], **d}
    init(json.dumps(d, ensure_ascii=False) + open(__file__, encoding="utf-8").read())
    mk = lambda path: SimpleDocTemplate(path, pagesize=A4, leftMargin=20*mm, rightMargin=20*mm,
                                        topMargin=18*mm, bottomMargin=20*mm, title="최종보고서")
    tmp = out + ".tmp.pdf"; doc = mk(tmp); doc.build(body_story(d)); n = doc.page; os.remove(tmp)
    mk(out).build(front_story(d, n) + body_story(d), onLaterPages=footer)

if __name__ == "__main__":
    build(json.load(open(sys.argv[1], encoding="utf-8")), sys.argv[2])
    print("생성 완료:", sys.argv[2])
